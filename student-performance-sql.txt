DROP TABLE IF EXISTS students;
CREATE TABLE students (
    student_id INTEGER PRIMARY KEY,
    name       VARCHAR(50) NOT NULL,
    department VARCHAR(30),
    semester   INTEGER,
    math       INTEGER,
    physics    INTEGER,
    programming INTEGER
);

SELECT * FROM students;
--1. Only EEE students
Select name, semester FROM students WHERE department = 'EEE';
--2. Programming above 80, best first
SELECT name, programming FROM students
WHERE programming > 80
ORDER BY programming DESC;
--4. Department-wise summary
SELECT department, COUNT(*) AS total_students, AVG(programming) AS avg_programming
FROM students
GROUP BY department;
--5. Grades
SELECT name,
       (math + physics + programming) / 3.0 AS average,
       CASE
           WHEN (math + physics + programming) / 3.0 >= 80 THEN 'A'
           WHEN (math + physics + programming) /3.0 >=65 THEN 'B'
           WHEN (math + physics + programming) /3.0 >= 50 THEN 'C'
           ELSE 'Fail'
         END AS grade
FROM students;
-- Top scorer in Math
SELECT name, math FROM students ORDER BY math DESC LIMIT 1;
-- Top scorer in Physics
SELECT name, physics FROM students ORDER BY physics DESC LIMIT 1;
-- Top scorer in Programming
SELECT name, programming FROM students ORDER BY programming DESC LIMIT 1;
